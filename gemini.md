```python
import google.generativeai as genai
from pydantic import BaseModel

# Initialize the Gemini API (Ensure GEMINI_API_KEY is set in Render's Environment Variables)
genai.configure(api_key=os.getenv("GEMINI_API_KEY"))

# Define what data the frontend will send
class ReviewRequest(BaseModel):
    topic: str
    target_demographic: str = "India" # Defaulting to your primary focus
    depth: str = "comprehensive"

@app.post("/api/literature-review")
async def generate_literature_review(request_data: ReviewRequest, _=Depends(verify_token)):
    try:
        # We use the pro model for complex synthesis like literature reviews
        model = genai.GenerativeModel('gemini-1.5-pro')
        
        prompt = f"""
        Conduct a {request_data.depth} literature review on the topic of: '{request_data.topic}'.
        Focus specifically on research, data, and context relevant to {request_data.target_demographic}.
        Structure the output with clear headings, a synthesis of current trends, and notable gaps in the research.
        """
        
        # Async call to the AI so the server doesn't block other users
        response = await model.generate_content_async(prompt)
        
        return {"status": "success", "review": response.text}
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"AI Engine Error: {str(e)}")
```

### **Phase 2: The Frontend Trigger**

Now, inside your Level/Session templates (e.g., `app/templates/level1/session1/index.html`), you need a way for the user to trigger this API and display the result. 

Because we are keeping things light and avoiding bloated legacy code, a simple vanilla JavaScript `fetch` call is perfect.

**Update `index.html` (Example):**
```html
<h2>AI Literature Review Generator</h2>

<input type="text" id="topicInput" placeholder="Enter research topic...">
<button onclick="fetchReview()">Generate Review</button>

<div id="loading" style="display:none;">Synthesizing research... this may take a moment.</div>
<div id="reviewOutput" style="margin-top: 20px; white-space: pre-wrap;"></div>

<script>
async function fetchReview() {
    const topic = document.getElementById('topicInput').value;
    const outputDiv = document.getElementById('reviewOutput');
    const loadingDiv = document.getElementById('loading');
    
    if (!topic) return alert("Please enter a topic.");

    // UI Updates
    outputDiv.innerHTML = "";
    loadingDiv.style.display = "block";

    try {
        const response = await fetch('/api/literature-review', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ 
                topic: topic, 
                target_demographic: "India" // Can be made dynamic later
            })
        });

        const data = await response.json();
        
        if (response.ok) {
            outputDiv.innerHTML = data.review; // In production, consider a Markdown parser here
        } else {
            outputDiv.innerHTML = "Error: " + data.detail;
        }
    } catch (error) {
        outputDiv.innerHTML = "Network error occurred.";
    } finally {
        loadingDiv.style.display = "none";
    }
}
</script>
```

### **The Architecture Explained**
1. **The Gate:** The `_=Depends(verify_token)` ensures no one can hit your AI endpoint (and cost you API credits) unless they've authenticated into the session.
2. **The Default Context:** By structuring the Pydantic model (`ReviewRequest`) with a default demographic, you guarantee the AI grounds its research accurately for your target audience without the user having to specify it every time.
3. **The Async Flow:** Using `generate_content_async` is the secret sauce. Literature reviews take a few seconds to generate. If this were synchronous, a single user requesting a review would freeze the entire Flask/FastAPI server for everyone else until it finished.

```bash
#!/bin/bash

echo "🔐 Installing Auth System..."

mkdir -p app/core

cat <<EOF > app/core/auth.py
from fastapi import Request, HTTPException, Depends
from passlib.context import CryptContext
import secrets

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# In-memory store (upgrade to DB later)
users = {
    "admin": pwd_context.hash("admin123")
}

active_sessions = {}

def verify_password(plain, hashed):
    return pwd_context.verify(plain, hashed)

def login_user(username: str, password: str):
    if username in users and verify_password(password, users[username]):
        token = secrets.token_urlsafe(16)
        active_sessions[token] = username
        return token
    return None

def get_current_user(request: Request):
    token = request.cookies.get("session_token")
    if token not in active_sessions:
        raise HTTPException(status_code=401, detail="Unauthorized")
    return active_sessions[token]
EOF

# Update requirements
echo "passlib[bcrypt]" >> requirements.txt

echo "🧩 Updating main.py..."

cat <<'EOF' > app/main.py
import os
from fastapi import FastAPI, Request, Form, Depends

app = FastAPI()
templates = Jinja2Templates(directory="app/templates")

@app.get("/", response_class=HTMLResponse)
async def home():
    return """
    <h2>Login</h2>
    <form method="post" action="/login">
        <input name="username" placeholder="user"/>
        <input name="password" type="password"/>
        <button>Login</button>
    </form>
    """

@app.post("/login")
async def login(username: str = Form(...), password: str = Form(...)):

    if not token:
        return RedirectResponse("/", status_code=302)

    res = RedirectResponse("/dashboard", status_code=302)

    return res

@app.get("/dashboard", response_class=HTMLResponse)
async def dashboard(user=Depends(get_current_user)):
```

```bash
#!/bin/bash

echo "⚡ Installing Streaming AI..."

cat <<'EOF' >> app/main.py

from fastapi.responses import StreamingResponse
import asyncio

async def fake_stream(text):
    for word in text.split():
        yield word + " "
        await asyncio.sleep(0.05)

@app.get("/api/stream")
async def stream(prompt: str):
    async def generator():
        async for chunk in fake_stream(f"Streaming response for: {prompt}"):
            yield chunk
    return StreamingResponse(generator(), media_type="text/plain")
EOF

echo "🧩 Updating frontend..."

cat <<EOF > app/static/js/stream.js
async function streamAI(){
  const input = document.getElementById("topic").value;
  const output = document.getElementById("out");

  output.innerText = "";

  const res = await fetch("/api/stream?prompt=" + input);
  const reader = res.body.getReader();
  const decoder = new TextDecoder();

  while(true){
    const {done, value} = await reader.read();
    if(done) break;
    output.innerText += decoder.decode(value);
  }
}
EOF

# Update template button
cat <<EOF > app/templates/session.html
<h1>Level {{l}} Session {{s}}</h1>

<input id="topic" placeholder="Enter topic"/>
<button onclick="streamAI()">Stream AI</button>

<pre id="out"></pre>

<script src="/static/js/stream.js"></script>
EOF

echo "✅ Streaming enabled"
```
