import numpy as np
import matplotlib.pyplot as plt

angle = float(input('Enter angle (deg): '))
rad = np.deg2rad(angle)
t = np.linspace(0, 10, 100)
x = 100 * np.cos(rad) * t
y = 100 * np.sin(rad) * t - 0.5 * 9.81 * t**2
plt.plot(x, y)
plt.title('Full Trajectory')
plt.xlabel('Distance (m)')
plt.ylabel('Height (m)')
plt.grid(True)
plt.show()
