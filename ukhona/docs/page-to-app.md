```python
import google.generativeai as genai
from pydantic import BaseModel

# Initialize the Gemini API (Ensure GEMINI_API_KEY is set in Render's Environment Variables)
genai.configure(api_key=os.getenv("GEMINI_API_KEY"))

# Define what data the frontend will send
class ReviewRequest(BaseModel):
    topic: str
    target_demographic: str = "India" # Defaulting to your primary focus
    depth: str = "comprehensive"

@app.post("/api/literature-review")
async def generate_literature_review(request_data: ReviewRequest, _=Depends(verify_token)):
    try:
        # We use the pro model for complex synthesis like literature reviews
        model = genai.GenerativeModel('gemini-1.5-pro')
        
        prompt = f"""
        Conduct a {request_data.depth} literature review on the topic of: '{request_data.topic}'.
        Focus specifically on research, data, and context relevant to {request_data.target_demographic}.
        Structure the output with clear headings, a synthesis of current trends, and notable gaps in the research.
        """
        
        # Async call to the AI so the server doesn't block other users
        response = await model.generate_content_async(prompt)
        
        return {"status": "success", "review": response.text}
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"AI Engine Error: {str(e)}")
```

### **Phase 2: The Frontend Trigger**

Now, inside your Level/Session templates (e.g., `app/templates/level1/session1/index.html`), you need a way for the user to trigger this API and display the result. 

Because we are keeping things light and avoiding bloated legacy code, a simple vanilla JavaScript `fetch` call is perfect.

**Update `index.html` (Example):**
```html
<h2>AI Literature Review Generator</h2>

<input type="text" id="topicInput" placeholder="Enter research topic...">
<button onclick="fetchReview()">Generate Review</button>

<div id="loading" style="display:none;">Synthesizing research... this may take a moment.</div>
<div id="reviewOutput" style="margin-top: 20px; white-space: pre-wrap;"></div>

<script>
async function fetchReview() {
    const topic = document.getElementById('topicInput').value;
    const outputDiv = document.getElementById('reviewOutput');
    const loadingDiv = document.getElementById('loading');
    
    if (!topic) return alert("Please enter a topic.");

    // UI Updates
    outputDiv.innerHTML = "";
    loadingDiv.style.display = "block";

    try {
        const response = await fetch('/api/literature-review', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ 
                topic: topic, 
                target_demographic: "India" // Can be made dynamic later
            })
        });

        const data = await response.json();
        
        if (response.ok) {
            outputDiv.innerHTML = data.review; // In production, consider a Markdown parser here
        } else {
            outputDiv.innerHTML = "Error: " + data.detail;
        }
    } catch (error) {
        outputDiv.innerHTML = "Network error occurred.";
    } finally {
        loadingDiv.style.display = "none";
    }
}
</script>
```

### **The Architecture Explained**
1. **The Gate:** The `_=Depends(verify_token)` ensures no one can hit your AI endpoint (and cost you API credits) unless they've authenticated into the session.
2. **The Default Context:** By structuring the Pydantic model (`ReviewRequest`) with a default demographic, you guarantee the AI grounds its research accurately for your target audience without the user having to specify it every time.
3. **The Async Flow:** Using `generate_content_async` is the secret sauce. Literature reviews take a few seconds to generate. If this were synchronous, a single user requesting a review would freeze the entire Flask/FastAPI server for everyone else until it finished.

```js
/**
 * Ukubona × WHO India — Anthropic API Proxy
 * Cloudflare Worker
 *
 * What this does:
 *   - Holds your Anthropic API key server-side (never exposed to the browser)
 *   - Optionally checks a shared password cookie before forwarding requests
 *   - Rate-limits to 20 requests per IP per hour (protects your credits)
 *   - Forwards only to api.anthropic.com/v1/messages (no other routes)
 *   - Returns CORS headers so your GitHub Pages HTML can call it
 *
 * Deploy in 3 minutes — see deploy.sh
 */

// ── Rate limit store (in-memory, resets per Worker instance)
// For production with many users, upgrade to Cloudflare KV.
const rateLimitStore = new Map();
const RATE_LIMIT = 20;          // max requests per window
const WINDOW_MS  = 60 * 60 * 1000; // 1 hour

function checkRateLimit(ip) {
  const now   = Date.now();
  const entry = rateLimitStore.get(ip);

  if (!entry || now - entry.start > WINDOW_MS) {
    rateLimitStore.set(ip, { count: 1, start: now });
    return true;
  }
  if (entry.count >= RATE_LIMIT) return false;
  entry.count++;
  return true;
}

// ── CORS headers — restrict to your GitHub Pages domain in production
function corsHeaders(origin) {
  // Change this to your exact domain once live, e.g.:
  //   "https://ukubona-llc.github.io"
  const allowed = [
    "https://ukubona-llc.github.io",
    "http://localhost",        // local dev
    "http://127.0.0.1",
  ];
  const allow = allowed.includes(origin) ? origin : allowed[0];
  return {
    "Access-Control-Allow-Origin":  allow,
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Access-Control-Max-Age":       "86400",
  };
}

export default {
  async fetch(request, env) {
    const origin = request.headers.get("Origin") || "";
    const cors   = corsHeaders(origin);

    // ── Preflight
    if (request.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: cors });
    }

    // ── Only allow POST to /v1/messages
    if (request.method !== "POST") {
      return new Response("Method not allowed", { status: 405, headers: cors });
    }

    // ── Optional password gate
    // Set PASSWORD secret in Cloudflare dashboard (or leave blank to disable).
    // If set, the browser must send:  Authorization: Bearer <PASSWORD>
    if (env.PASSWORD) {
      const auth  = request.headers.get("Authorization") || "";
      const token = auth.replace("Bearer ", "").trim();
      if (token !== env.PASSWORD) {
        return new Response(
          JSON.stringify({ error: "Unauthorized" }),
          { status: 401, headers: { ...cors, "Content-Type": "application/json" } }
        );
      }
    }

    // ── Rate limit
    const ip = request.headers.get("CF-Connecting-IP") || "unknown";
    if (!checkRateLimit(ip)) {
      return new Response(
        JSON.stringify({ error: "Rate limit exceeded. Please try again in an hour." }),
        { status: 429, headers: { ...cors, "Content-Type": "application/json" } }
      );
    }

    // ── Parse and validate body
    let body;
    try {
      body = await request.json();
    } catch {
      return new Response(
        JSON.stringify({ error: "Invalid JSON body" }),
        { status: 400, headers: { ...cors, "Content-Type": "application/json" } }
      );
    }

    // Safety: enforce model to claude-sonnet only (prevents abuse)
    const allowedModels = [
      "claude-sonnet-4-20250514",
      "claude-haiku-4-5-20251001",
    ];
    if (!allowedModels.includes(body.model)) {
      body.model = "claude-sonnet-4-20250514";
    }

    // Safety: cap max_tokens
    if (!body.max_tokens || body.max_tokens > 1000) {
      body.max_tokens = 1000;
    }

    // ── Forward to Anthropic
    let anthropicResponse;
    try {
      anthropicResponse = await fetch("https://api.anthropic.com/v1/messages", {
        method:  "POST",
        headers: {
          "Content-Type":      "application/json",
          "x-api-key":         env.ANTHROPIC_API_KEY,
          "anthropic-version": "2023-06-01",
        },
        body: JSON.stringify(body),
      });
    } catch (err) {
      return new Response(
        JSON.stringify({ error: "Failed to reach Anthropic API", detail: err.message }),
        { status: 502, headers: { ...cors, "Content-Type": "application/json" } }
      );
    }

    // ── Stream the response back (preserves streaming if used)
    const responseBody    = await anthropicResponse.text();
    const responseHeaders = {
      ...cors,
      "Content-Type": anthropicResponse.headers.get("Content-Type") || "application/json",
    };

    return new Response(responseBody, {
      status:  anthropicResponse.status,
      headers: responseHeaders,
    });
  },
};
```

```sh
#!/bin/bash
# ============================================================
# patch_sessions.sh — Update all session HTML files to use
# the Cloudflare Worker proxy instead of Anthropic directly
#
# Run AFTER deploy.sh, from inside your who-i repo root:
#   bash patch_sessions.sh
# ============================================================

set -e
clear

echo "======================================================"
echo " 🔧 Session HTML Patcher"
echo "======================================================"
echo ""

read -p "Enter your worker URL (e.g. https://ukubona-who-proxy.abc123.workers.dev): " WORKER_URL
if [ -z "$WORKER_URL" ]; then
  echo "❌ Worker URL is required."
  exit 1
fi

# Strip trailing slash if present
WORKER_URL="${WORKER_URL%/}"
PROXY_ENDPOINT="${WORKER_URL}/v1/messages"

read -p "Enter your access password (press Enter if none): " CLASS_PASSWORD
echo ""

# ── Find all session HTML files ───────────────────────────
SESSION_FILES=$(find ukhona/html -name "session*.html" 2>/dev/null)
COUNT=0

for file in $SESSION_FILES; do
  echo "Patching: $file"

  # Replace the Anthropic endpoint with the proxy
  sed -i "s|https://api.anthropic.com/v1/messages|${PROXY_ENDPOINT}|g" "$file"

  # Remove the x-api-key header line (proxy adds it)
  sed -i '/"x-api-key"/d' "$file"

  # If a password was set, inject Authorization header after Content-Type
  if [ -n "$CLASS_PASSWORD" ]; then
    sed -i "s|\"Content-Type\": \"application/json\"|\"Content-Type\": \"application/json\",\n          \"Authorization\": \"Bearer ${CLASS_PASSWORD}\"|g" "$file"
  fi

  COUNT=$((COUNT + 1))
done

echo ""
echo "======================================================"
echo " ✅ Patched $COUNT session files"
echo "======================================================"
echo ""
echo "Commit and push:"
echo "  git add ukhona/html"
echo "  git commit -m 'chore: route AI calls through Cloudflare proxy'"
echo "  git push"
echo ""
echo "GitHub Pages will update within ~60 seconds."
echo "======================================================"
```
